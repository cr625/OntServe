{# Entity Display Macros - Simple property rendering for all RDF data #} {# Uses format_key filter from app.py for camelCase to Title Case #} {# Format a value for display - handles URIs, lists, etc #} {% macro format_value(value) %} {%- if value is string -%} {%- if value.startswith('http://') or value.startswith('https://') -%} {%- set short = value.split('#')[-1].split('/')[-1] -%} {{ short }} {%- else -%} {{ value }} {%- endif -%} {%- elif value is iterable and value is not string -%} {%- for v in value -%} {{ format_value(v) }}{% if not loop.last %}, {% endif %} {%- endfor -%} {%- else -%} {{ value }} {%- endif -%} {% endmacro %} {# Main macro to render all entity properties #} {% macro render_entity_properties(entity) %} {% if entity.properties %} {% set props = entity.properties if entity.properties is mapping else {} %} {% if props|length > 0 %}
{% for key, value in props.items() %} {% if value is not none and value != '' %}
{{ key|format_key }}: {{ format_value(value) }}
{% endif %} {% endfor %}
{% endif %} {% endif %} {% endmacro %} {# Compact version for list views - just first property #} {% macro render_entity_properties_compact(entity) %} {% if entity.properties %} {% set props = entity.properties if entity.properties is mapping else {} %} {% if props|length > 0 %}
{% for key, value in props.items() %} {% if loop.first and value is not none %} {{ format_value(value)|truncate(100) }} {% endif %} {% endfor %}
{% endif %} {% endif %} {% endmacro %}
