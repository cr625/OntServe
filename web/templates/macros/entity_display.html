{# Entity Display Macros - Simple property rendering for all RDF data #} {# Uses format_key filter from app.py for camelCase to Title Case #} {# Metadata property keys to hide from card display (case-insensitive match). These are extraction provenance / internal tracking, not semantic content. Full details remain available on the entity detail page. #} {% set HIDDEN_PROPERTY_KEYS = [ 'generatedattime', 'wasgeneratedby', 'wasattributedto', 'conceptcategory', 'content_hash', 'discoveredincase', 'discoveredinpass', 'discoveredinsection', 'firstdiscoveredat', 'firstdiscoveredincase', 'textreferences', 'sourcetext', 'importance', 'confidence', 'attributes', 'relationships' ] %} {# Format a value for display - handles URIs, lists, etc #} {% macro format_value(value) %} {%- if value is string -%} {%- if value.startswith('http://') or value.startswith('https://') -%} {%- set short = value.split('#')[-1].split('/')[-1] -%} {{ short | split_pascal }} {%- else -%} {{ value }} {%- endif -%} {%- elif value is iterable and value is not string -%} {%- for v in value -%} {{ format_value(v) }}{% if not loop.last %}, {% endif %} {%- endfor -%} {%- else -%} {{ value }} {%- endif -%} {% endmacro %} {# Main macro to render entity properties (filtered) #} {% macro render_entity_properties(entity) %} {% if entity.properties %} {% set props = entity.properties if entity.properties is mapping else {} %} {% if props|length > 0 %}
{% for key, value in props.items() %} {% if value is not none and value != '' and key|lower not in HIDDEN_PROPERTY_KEYS %}
{{ key|format_key }}: {{ format_value(value) }}
{% endif %} {% endfor %}
{% endif %} {% endif %} {% endmacro %} {# Full properties (no filtering) for entity detail page #} {% macro render_entity_properties_full(entity) %} {% if entity.properties %} {% set props = entity.properties if entity.properties is mapping else {} %} {% if props|length > 0 %}
{% for key, value in props.items() %} {% if value is not none and value != '' %}
{{ key|format_key }}: {{ format_value(value) }}
{% endif %} {% endfor %}
{% endif %} {% endif %} {% endmacro %} {# Compact version for list views - just first property #} {% macro render_entity_properties_compact(entity) %} {% if entity.properties %} {% set props = entity.properties if entity.properties is mapping else {} %} {% if props|length > 0 %}
{% for key, value in props.items() %} {% if loop.first and value is not none %} {{ format_value(value)|truncate(100) }} {% endif %} {% endfor %}
{% endif %} {% endif %} {% endmacro %}
